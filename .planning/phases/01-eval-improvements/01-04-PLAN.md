    export function median(values: number[]): number | null;

    export interface RunMetrics {
      dir: string; run_id: string;               // run_id = basename(dir)
      harness_version: number;                   // grades.harness_version ?? 1
      regrade_of: string | null;
      models: { route: string | null; generation: string | null; embedding: string | null; judge: string; judge_route: string | null };
      total: number; graded: number;
      criteria: Record<Criterion, number | null>;               // mean score over graded records
      hallucinations: number;
      responses_with_unsupported_claims: number;               // grades with invented_claims.length > 0
      unsupported_claims_total: number;                        // sum of invented_claims.length
      claims_by_kind: Record<ClaimKind, number> | null;        // null if any graded record lacks `claims`
      responses_by_kind: Record<ClaimKind, number> | null;
      recall: { hit: number; total: number; pct: number | null };   // same rule as report.ts today
      latency_s: { median: number | null; min: number | null; max: number | null }; // ok records only, seconds, 1 dp
      metrics_not_in_context: number;   // RECOMPUTED with runAutomatedChecks on responses (current normalisation), not read from stored grades
      strict_fallback: { pass: number; of: number };          // same row filter as report.ts "Fallback behaviour"
      strategy_lead: { expected: number; offered_when_expected: number }; // recomputed via runAutomatedChecks
    }
    export function computeRunMetrics(dir: string): RunMetrics; // reads responses.json + grades.json (grades optional: score fields null/0 when absent)

    export interface MetricDef { key: string; label: string; better: 'higher' | 'lower' | 'none'; unit: 'score' | 'count' | 'pct' | 'seconds'; get: (m: RunMetrics) => number | null; newHarnessOnly?: boolean }
    export const METRIC_DEFS: MetricDef[];
    /** Most recent dir in RESULTS_DIR whose responses.json has regrade_of === baselineId, or null. */
