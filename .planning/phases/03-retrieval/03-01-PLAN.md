    ```sql
    CREATE OR REPLACE FUNCTION hybrid_search_v2(
      query_embedding vector(1536),
      query_text TEXT,
      match_count INT DEFAULT 60,
      doc_types document_type[] DEFAULT NULL,
      boost_capability_slugs TEXT[] DEFAULT NULL,
      boost_industry_slugs TEXT[] DEFAULT NULL,
      include_company_info BOOLEAN DEFAULT FALSE,
      rrf_k INT DEFAULT 60,
      intent_boost FLOAT DEFAULT 0.0
    )
    RETURNS TABLE (
      chunk_id UUID, document_id UUID, document_title TEXT, document_type document_type,
      client_name TEXT, author TEXT, slug TEXT, vimeo_url TEXT, thumbnail_url TEXT, pdf_url TEXT, hero_video_url TEXT,
      content TEXT, chunk_type TEXT, metadata JSONB,
      semantic_score FLOAT, keyword_score FLOAT, semantic_rank INT, keyword_rank INT,
      intent_match BOOLEAN, combined_score FLOAT
    ) AS $$
    #variable_conflict use_column
    DECLARE
      or_query tsquery := NULLIF(
        regexp_replace(replace(plainto_tsquery('english', query_text)::text, ' & ', ' | '), ' <[0-9-]+> ', ' | ', 'g'), ''
      )::tsquery;
    BEGIN
      PERFORM set_config('ivfflat.probes', '100', true);
      RETURN QUERY
      WITH base AS ( ... exactly as the research sketch's base CTE, plus d.author AS d_author ... ),
      scored AS ( SELECT b.*, CASE WHEN or_query IS NOT NULL AND b.vec @@ or_query THEN ts_rank(b.vec, or_query) ELSE 0 END::FLOAT AS kw FROM base b ),
      ranked AS (
        SELECT s.*,
               rank() OVER (PARTITION BY s.d_type ORDER BY s.sem DESC)::INT AS srank,
               (CASE WHEN s.kw > 0 THEN rank() OVER (PARTITION BY s.d_type, (s.kw > 0) ORDER BY s.kw DESC) END)::INT AS krank
        FROM scored s
      ),
      fused AS (
        SELECT r.*, (1.0 / (rrf_k + r.srank) + COALESCE(1.0 / (rrf_k + r.krank), 0)
                     + CASE WHEN r.intent_hit THEN intent_boost ELSE 0 END)::FLOAT AS rrf
        FROM ranked r
      )
      SELECT f.c_id, f.d_id, f.d_title, f.d_type, f.d_client, f.d_author, f.d_slug, f.d_vimeo, f.d_thumb, f.d_pdf, f.d_hero,
             f.c_content, f.c_type, f.c_meta, f.sem, f.kw, f.srank, f.krank, f.intent_hit, f.rrf
      FROM fused f
      ORDER BY f.rrf DESC, f.c_id
      LIMIT match_count;
    END;
    $$ LANGUAGE plpgsql STABLE;

    GRANT EXECUTE ON FUNCTION hybrid_search_v2(vector, text, int, document_type[], text[], text[], boolean, int, float)
      TO anon, authenticated, service_role;
    ```
    The `base` CTE (copy from research Pattern 1): semantic `(1 - (cc.embedding <=> query_embedding))::FLOAT AS sem` computed for EVERY row (no `ORDER BY embedding <=> … LIMIT`, which is the lossy ANN plan); weighted vector
    `setweight(to_tsvector('english', d.title),'A') || setweight(to_tsvector('english', COALESCE(d.client_name,'')),'A') || setweight(to_tsvector('english', COALESCE(tags.names,'')),'B') || setweight(to_tsvector('english', cc.content),'D') AS vec`;
    `intent_hit = COALESCE(tags.cap_slugs && boost_capability_slugs, FALSE) OR COALESCE(tags.ind_slugs && boost_industry_slugs, FALSE)`;
    the LEFT JOIN LATERAL `tags` subquery aggregating capability and industry `name`s and slugs; and the WHERE clause:
    `cc.embedding IS NOT NULL AND (doc_types IS NULL OR d.doc_type = ANY(doc_types)) AND COALESCE(cc.metadata->>'article_class','') <> 'excluded' AND (include_company_info OR COALESCE(cc.metadata->>'article_class','') <> 'company_info')` (D-16 and D-15).
    Every column reference inside the CTEs must be table-qualified (cc., d., x., s., r., f.) so `#variable_conflict use_column` can't capture an output-column name.

    Optional local syntax check: `/opt/local/bin/psql` is PG 16.11 without pgvector, so only check the OR-query expression there:
    `SELECT regexp_replace(replace(plainto_tsquery('english','messaging house + marketecture? go-to-market')::text,' & ',' | '),' <[0-9-]+> ',' | ','g')::tsquery;` should produce only `|` operators (no `&`, `<->` or `<N>`). Record the output in the SUMMARY. Skip it if no local server is running.
