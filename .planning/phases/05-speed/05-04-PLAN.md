    ```typescript
      | { type: 'hero'; title?: string; subtitle?: string; challengeSummary?: string; backgroundVariant?: 'dark' | 'light' | 'gradient' } // cumulative

      | { type: 'component'; index: number; component: string; props: Record<string, unknown> }
      | { type: 'narrative'; text: string } // delta
      | { type: 'abandoned'; reason: string };
    export class StreamLayoutParser {
      push(text: string): ParserEvent[];
      end(): ParserEvent[];       // flushes held-back narrative; never throws
      get fullText(): string;     // all pushed text concatenated
    }
    ```
