    2. Create `src/lib/provider-health.ts` (server-only; no Next imports so eval can import it):
       ```typescript
       export type HealthStatus = 'ok' | 'warn' | 'fail';
       export interface ProviderHealth {
         status: HealthStatus; reason: string; checkedAt: string;
         route: ProviderRoute; models: { generation: string; embedding: string };
         key: { present: boolean; valid: boolean | null };
         credit: { limit: number | null; remaining: number | null; usage: number | null };
         httpStatus: number | null;
       }
       export async function checkProviderHealth(opts?: { timeoutMs?: number; fetchImpl?: typeof fetch; warnBelowUsd?: number }): Promise<ProviderHealth>;
       export function runStartupHealthCheck(): void;
       ```
       Logic order: route !== 'openrouter' -> fail ("Model calls are on the direct route, not OpenRouter; set OPENROUTER_API_KEY"); missing key -> fail; GET /api/v1/key with AbortController timeout (default CONCIERGE_CONFIG.health.requestTimeoutMs); 401/403 -> fail ("OpenRouter key is invalid (HTTP n)"); other non-ok -> warn ("OpenRouter key check returned HTTP n"); timeout -> warn ("OpenRouter key check timed out after n ms"); network error -> warn ("Could not reach OpenRouter: <message>"); spendable = minimum of the non-null numbers among `limit_remaining` and `effective_budget.remaining_usd`; null -> warn ("credit unknown: key has no limit; the account balance needs a management key"); <= 0 -> fail ("OpenRouter credit is used up ($x.xx left)"); < warnBelowUsd -> warn ("OpenRouter credit $x.xx is below $5.00"); else ok ("OpenRouter route, key valid, $x.xx credit left"). Never throws.
       `runStartupHealthCheck()`: `void checkProviderHealth({ timeoutMs: CONCIERGE_CONFIG.health.startupTimeoutMs }).then(h => h.status === 'ok' ? console.info(`[provider-health] ok: ${h.reason}`) : console.warn(`[provider-health] ${h.status.toUpperCase()}: ${h.reason} (route ${h.route}, model ${h.models.generation})`)).catch(() => {})`.
